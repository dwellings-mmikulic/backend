# QR Bumper + `ip=` Channel Switch Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every linear channel airs a generated QR bumper clip every 5 minutes, and a platform that fetches `/channels/live.m3u8?ip=<viewer ip>` gets that household's personal feed once the household exists, with continuous HLS counters.

**Architecture:** The bumper is a lineup item from its own `bumpers` table, referenced in `channel_lineups.item_ids` as the negated bumper id; the lineup builder inserts it after every `BumperInterval` of content; `ClipsByID` resolves mixed ids so playlists, personal feeds and the chain math do not change; the EPG skips bumper items. The channel `live` handler, given a public `ip=` and an existing household, serves an *anchored* personal playlist: a stored per-(household, channel) offset pair makes the personal counters continue the channel's, exactly like a span splice. A new `internal/bumper` package renders the clip with ffmpeg; `cmd/make-bumper` renders, segments, uploads and registers it.

**Tech Stack:** Go 1.23+, pgx, ffmpeg (Alpine, DejaVu font), `internal/hls`, `internal/qrcode`, `internal/bunny`.

**Spec:** `docs/superpowers/specs/2026-10-13-qr-bumper-design.md`

## Global Constraints

- Encode profile for the bumper must match listings: `libx264`, `yuv420p`, `-crf 20`, `-preset veryfast`, `-r 30` (`config.MinVideoFPS`), AAC `160k`, `+faststart`; the segmenter maps `0:a:0` so the bumper always has an audio track.
- Bumper segments must each be ≤ `hls.TargetDuration` (10 s): force keyframes every 3 s (`-g 90 -keyint_min 90 -sc_threshold 0` at 30 fps).
- `channel_lineups.item_ids` schema is unchanged; a bumper item is the negated bumper id; `bumperItem`/`isBumper`/`bumperID` are the only code that knows this.
- With no bumper row, or `Options.BumperInterval == 0`, lineups are byte-identical to today.
- Rows in `bumpers` and `channel_anchors` are never deleted.
- `household` columns are `BYTEA` (viewer.ID bytes), as in `household_spans`.
- `ip=` is still ignored on every `/feed/` route (unchanged).
- Commits end with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.

## Review Focus

1. **A bumper row that disappears or disagrees with a version** (segment count differs): playlist must fail with `InconsistentLineupError`, never emit a lineup with shifted sequence numbers. Test in Task 3 (`TestWindow_BumperMismatchIsInconsistent`).
2. **A version with a single clip, or clips shorter than the interval in total**: no bumper inserted, never a bumper last. Test in Task 3 (`TestBuildItems_BumperNeverFirstOrLast`).
3. **Anchored playlist once the switch is older than the window** (channel part empty): still ≥ 6 segments / 40 s and counters monotonic across the whole poll series. Test in Task 6 (`TestAnchoredPlaylist_SwitchesWithMonotonicCounters` polls 3 minutes).
4. **`ip=` with a private/CGNAT/garbage address**: channel served unchanged, no household lookup. Test in Task 7 (`TestHandler_LiveIgnoresNonPublicIP`).
5. **EPG block whose first item is a bumper**: count and price range come from listings only, the title from the version scope. Test in Task 4 (`TestEPG_SkipsBumpers`).
6. **Anchor created while the household has a pending span** starting right after `S`: the first personal segment is the pending span's, so counters stay monotonic. Test in Task 6 (`TestAnchoredPlaylist_PendingSpanAtSwitch`).

```go
// A bumper is a short clip of our own (the QR invitation, see
// docs/superpowers/specs/2026-10-13-qr-bumper-design.md) that the lineup
// builder inserts between listings. Lineups reference clips by video_hls id

// only code that knows the sign convention.

// bumperItem is the lineup item id of bumper id.
func bumperItem(id int64) int64 { return -id }

// isBumper reports whether a lineup item references a bumper.
func isBumper(item int64) bool { return item < 0 }

// bumperID is the bumpers id behind a bumper lineup item.
func bumperID(item int64) int64 { return -item }

// ref is the bumper as the lineup builder sees it.
func (b *Bumper) ref() ClipRef {
	return ClipRef{ID: bumperItem(b.ID), TotalMS: b.TotalMS, Segments: len(b.SegmentMS)}
}
```

```go
package linear

import (
	"context"
	"errors"
	"fmt"

	"github.com/jackc/pgx/v5"
)

const bumperColumns = `id, name, content_hash, base_url, segment_ms, total_ms, created_at`

func scanBumper(row pgx.Row) (*Bumper, error) {
	var b Bumper
	var ms []int32
	if err := row.Scan(&b.ID, &b.Name, &b.ContentHash, &b.BaseURL, &ms, &b.TotalMS, &b.CreatedAt); err != nil {
		return nil, err
	}
	b.SegmentMS = ints(ms)
	return &b, nil
}

// InsertBumper stores b unless a bumper with its content hash exists; it
// returns the stored row's id either way, and whether this call created it.
func (r *Repository) InsertBumper(ctx context.Context, b *Bumper) (int64, bool, error) {
	const q = `
INSERT INTO bumpers (name, content_hash, base_url, segment_ms, total_ms)
VALUES ($1, $2, $3, $4, $5)
ON CONFLICT (content_hash) DO NOTHING
RETURNING id`
	var id int64
	err := r.pool.QueryRow(ctx, q, b.Name, b.ContentHash, b.BaseURL, int32s(b.SegmentMS), b.TotalMS).Scan(&id)
	if err == nil {
		return id, true, nil
	}
	if !errors.Is(err, pgx.ErrNoRows) {
		return 0, false, fmt.Errorf("insert bumper: %w", err)
	}
	existing, err := r.BumperByHash(ctx, b.ContentHash)
	if err != nil {
		return 0, false, err
	}
	if existing == nil {
		return 0, false, fmt.Errorf("insert bumper: conflict on %s but no row", b.ContentHash)
	}
	return existing.ID, false, nil
}

// BumperByHash returns the bumper with content hash hash, or nil.
func (r *Repository) BumperByHash(ctx context.Context, hash string) (*Bumper, error) {
	b, err := scanBumper(r.pool.QueryRow(ctx, `SELECT `+bumperColumns+` FROM bumpers WHERE content_hash = $1`, hash))
	if errors.Is(err, pgx.ErrNoRows) {
		return nil, nil
	}
	if err != nil {
		return nil, fmt.Errorf("bumper by hash: %w", err)
	}
	return b, nil
}

// CurrentBumper implements Store: the newest row.
func (r *Repository) CurrentBumper(ctx context.Context) (*Bumper, error) {
	b, err := scanBumper(r.pool.QueryRow(ctx, `SELECT `+bumperColumns+` FROM bumpers ORDER BY id DESC LIMIT 1`))
	if errors.Is(err, pgx.ErrNoRows) {
		return nil, nil
	}
	if err != nil {
		return nil, fmt.Errorf("current bumper: %w", err)
	}
	return b, nil
}
```

```go
// ClipsByID implements Store. Positive ids are video_hls rows, negative ones
// bumpers (bumperID); each entry is keyed by the id as given.
func (r *Repository) ClipsByID(ctx context.Context, ids []int64) (map[int64]ClipSegments, error) {
	var clipIDs, bumperIDs []int64
	for _, id := range ids {
		if isBumper(id) {
			bumperIDs = append(bumperIDs, bumperID(id))
		} else {
			clipIDs = append(clipIDs, id)
		}
	}
	out := make(map[int64]ClipSegments, len(ids))
	scan := func(q string, in []int64, key func(int64) int64) error {
		if len(in) == 0 {
			return nil
		}
		rows, err := r.pool.Query(ctx, q, in)
		if err != nil {
			return fmt.Errorf("clips by id: %w", err)
		}
		defer rows.Close()
		for rows.Next() {
			var c ClipSegments
			var ms []int32
			if err := rows.Scan(&c.ID, &c.BaseURL, &ms); err != nil {
				return fmt.Errorf("scan clip segments: %w", err)
			}
			c.ID = key(c.ID)
			c.SegmentMS = ints(ms)
			out[c.ID] = c
		}
		return rows.Err()
	}
	if err := scan(`SELECT id, base_url, segment_ms FROM video_hls WHERE id = ANY($1)`, clipIDs, func(id int64) int64 { return id }); err != nil {
		return nil, err
	}
	if err := scan(`SELECT id, base_url, segment_ms FROM bumpers WHERE id = ANY($1)`, bumperIDs, bumperItem); err != nil {
		return nil, err
	}
	return out, nil
}
```

```go
// buildItems picks a version's air order: a seeded shuffle of clips (sorted
// by id first so the input order is irrelevant), cut to maxMS. A lineup always
// has at least one clip, even if that clip alone exceeds maxMS. With a bumper
// and a positive interval, the bumper is then inserted in front of every clip
// that would otherwise start interval or more after the previous bumper ended
// (for the first one: after the version's start). It is never the first item
// and never the last: a version starts and ends with content.
func buildItems(clips []ClipRef, bumper *ClipRef, interval time.Duration, seed int64, maxMS int) (ids []int64, ms []int, segs []int) {
	order := make([]ClipRef, len(clips))
	copy(order, clips)
	sort.Slice(order, func(i, j int) bool { return order[i].ID < order[j].ID })
	r := rand.New(rand.NewSource(seed))
	r.Shuffle(len(order), func(i, j int) { order[i], order[j] = order[j], order[i] })

	total := 0
	var items []ClipRef
	for _, c := range order {
		if len(items) > 0 && total+c.TotalMS > maxMS {
			break
		}
		items = append(items, c)
		total += c.TotalMS
	}
	if bumper != nil && interval > 0 {
		items = withBumpers(items, *bumper, int(interval.Milliseconds()))
	}
	for _, c := range items {
		ids = append(ids, c.ID)
		ms = append(ms, c.TotalMS)
		segs = append(segs, c.Segments)
	}
	return ids, ms, segs
}

// withBumpers inserts bumper before every clip of items that would start
// intervalMS or more after the previous bumper's end (initially: after the
// start). The first item is always a clip, so is the last.
func withBumpers(items []ClipRef, bumper ClipRef, intervalMS int) []ClipRef {
	out := make([]ClipRef, 0, len(items)+len(items)/2)
	since := 0 // content ms since the last bumper ended
	for i, c := range items {
		if i > 0 && since >= intervalMS {
			out = append(out, bumper)
			since = 0
		}
		out = append(out, c)
		since += c.TotalMS
	}
	return out
}
```

```go
// personalSegments is the household's feed at now, untrimmed: the spans'
// contributions oldest first, each ended by now. airing is the span on air.
func (s *Service) personalSegments(ctx context.Context, h viewer.ID, now time.Time) ([]segment, Span, error) {
	spans, err := s.Household(ctx, h)
	if err != nil {
		return nil, Span{}, err
	}
	airing := spans[0]
	var segs []segment
	for k, sp := range spans {
		if sp.StartsAt.After(now) {
			break // not airing yet
		}
		airing = sp
		end := now
		if k+1 < len(spans) && spans[k+1].StartsAt.Before(end) {
			end = spans[k+1].StartsAt
		}
		if end.Before(now.Add(-lookback)) {
			continue
		}
		c, err := s.contribution(ctx, sp, end)
		if err != nil {
			return nil, Span{}, err
		}
		segs = append(segs, c...)
	}
	return segs, airing, nil
}

// PersonalPlaylist renders household h's live playlist at now. airing is the
// span on air at now.
func (s *Service) PersonalPlaylist(ctx context.Context, h viewer.ID, now time.Time) ([]byte, Span, error) {
	segs, airing, err := s.personalSegments(ctx, h, now)
	if err != nil {
		return nil, Span{}, err
	}
	if len(segs) == 0 {
		return nil, Span{}, ErrNoContent
	}
	var buf bytes.Buffer
	writePlaylist(&buf, liveWindow.trim(segs))
	return buf.Bytes(), airing, nil
}

// firstPersonalAtOrAfter is the first segment of the household's feed (spans,
// oldest first) that starts at or after t, relabelled with its span's
// offsets. It is the span airing at t's next segment — unless that segment
// would end after the following span starts, in which case (as in
// contribution) it never airs and the following span's first segment is the
// answer.
func (s *Service) firstPersonalAtOrAfter(ctx context.Context, spans []Span, t time.Time) (segment, error) {
	k := 0
	for k+1 < len(spans) && !spans[k+1].StartsAt.After(t) {
		k++
	}
	for ; k < len(spans); k++ {
		sp := spans[k]
		from := t
		if sp.StartsAt.After(from) {
			from = sp.StartsAt
		}
		g, err := s.firstAtOrAfter(ctx, sp.Scope, from)
		if err != nil {
			return segment{}, err
		}
		end := g.Start.Add(time.Duration(g.DurMS) * time.Millisecond)
		if k+1 < len(spans) && end.After(spans[k+1].StartsAt) {
			continue // cut off by the next span: it never airs
		}
		return shifted([]segment{g}, sp)[0], nil
	}
	return segment{}, fmt.Errorf("household %s: no segment at or after %s", spans[0].Household, t)
}
```

```go
// AnchoredPlaylist renders household h's feed as the continuation of
// channel sc, for a platform that fetches the channel with the viewer's ip=
// (see the design spec, "Anchoring"). The first call for (h, sc) creates
// the anchor: the switch point S is the end of the channel segment on air,
// and the offsets make the first personal segment after S carry the
// counters the channel's next segment would have. Every call then serves
// the channel's segments that ended by S (while S is still within the
// window) followed by the household's from S on, relabelled.
//
// ErrUnknownHousehold when h has no feed: the caller serves the channel.
func (s *Service) AnchoredPlaylist(ctx context.Context, h viewer.ID, sc Scope, now time.Time) ([]byte, error) {
	if err := s.checkArea(ctx, sc); err != nil {
		return nil, err
	}
	spans, err := s.Household(ctx, h)
	if err != nil {
		return nil, err
	}
	key := sc.Key()
	a, err := s.store.Anchor(ctx, h, key)
	if err != nil {
		return nil, err
	}
	if a == nil {
		if a, err = s.anchor(ctx, h, spans, key, now); err != nil {
			return nil, err
		}
	}
	var segs []segment
	if !a.StartsAt.Before(now.Add(-lookback)) {
		segs, err = s.channelWindow(ctx, key, a.StartsAt)
		if err != nil {
			return nil, err
		}
		segs = ended(segs, a.StartsAt)
	}
	personal, _, err := s.personalSegments(ctx, h, now)
	if err != nil {
		return nil, err
	}
	i := 0
	for i < len(personal) && personal[i].Start.Before(a.StartsAt) {
		i++
	}
	segs = append(segs, shifted(personal[i:], Span{N: 1, SeqOffset: a.SeqOffset, ItemOffset: a.ItemOffset})...)
	if len(segs) == 0 {
		return nil, ErrNoContent
	}
	var buf bytes.Buffer
	writePlaylist(&buf, liveWindow.trim(segs))
	return buf.Bytes(), nil
}

// anchor creates and stores h's anchor on channel key at now, then re-reads
// it so a concurrent creator's row wins.
func (s *Service) anchor(ctx context.Context, h viewer.ID, spans []Span, key string, now time.Time) (*Anchor, error) {
	S, err := s.endOfAiring(ctx, key, now)
	if err != nil {
		return nil, err
	}
	before, err := s.channelWindow(ctx, key, S)
	if err != nil {
		return nil, err
	}
	before = ended(before, S)
	if len(before) == 0 {
		return nil, fmt.Errorf("channel %s: nothing aired before %s", key, S)
	}
	last := before[len(before)-1]
	first, err := s.firstPersonalAtOrAfter(ctx, spans, S)
	if err != nil {
		return nil, err
	}
	a := &Anchor{Household: h, Channel: key, StartsAt: S,
		SeqOffset: last.Seq + 1 - first.Seq, ItemOffset: last.Item + 1 - first.Item, CreatedAt: now}
	if _, err := s.store.InsertAnchor(ctx, a); err != nil {
		return nil, err
	}
	stored, err := s.store.Anchor(ctx, h, key)
	if err != nil {
		return nil, err
	}
	if stored == nil {
		return nil, fmt.Errorf("household %s: anchor on %s vanished after insert", h, key)
	}
	s.log.Info("channel anchor created", "household", h.String(), "channel", key, "starts_at", S)
	return stored, nil
}
```

```go
// Defaults.
const (
	DefaultURL     = "https://dwellings.tv/tv/"
	DefaultCaption = "Looking for something more specific?\nCustomize your channel through the QR code below"
	DefaultSeconds = 12
	DefaultColor   = "0x101418"

	// renderVersion is mixed into the hash: bump it when the layout or the
	// encode changes so an old bumper is not mistaken for the new one.
	renderVersion = "b1"
	qrSize        = 560 // QR pixels; on a 600 px white card
	qrCard        = 600
)

// Spec is everything the render depends on.
type Spec struct {
	URL             string // what the QR encodes
	Caption         string // lines separated by "\n"
	Seconds         int
	Background      string // ffmpeg colour, used without BackgroundImage
	BackgroundImage string // optional 16:9 image path
	Music           string // optional audio path; silence without
	FontPath        string
	FPS             int
}

// Hash identifies the render: the inputs, the bytes of the files among
// them, and renderVersion.
func (s Spec) Hash() (string, error) {
	h := sha256.New()
	fmt.Fprintf(h, "%s|%s|%s|%d|%s|%s|%d|", renderVersion, s.URL, s.Caption, s.Seconds, s.Background, s.FontPath, s.FPS)
	for _, p := range []string{s.BackgroundImage, s.Music} {
		if p == "" {
			io.WriteString(h, "-|")
			continue
		}
		f, err := os.Open(p)
		if err != nil {
			return "", err
		}
		_, err = io.Copy(h, f)
		f.Close()
		if err != nil {
			return "", err
		}
		io.WriteString(h, "|")
	}
	return hex.EncodeToString(h.Sum(nil)), nil
}

// Render writes the bumper MP4 into workDir and returns its path.
func Render(ctx context.Context, s Spec, workDir string) (string, error) {
	qrPath := filepath.Join(workDir, "qr.png")
	if err := qrcode.WritePNG(s.URL, qrPath, qrSize); err != nil {
		return "", err
	}
	var captionFiles []string
	for i, line := range strings.Split(s.Caption, "\n") {
		p := filepath.Join(workDir, fmt.Sprintf("caption-%d.txt", i))
		if err := os.WriteFile(p, []byte(line), 0o644); err != nil {
			return "", err
		}
		captionFiles = append(captionFiles, p)
	}
	urlFile := filepath.Join(workDir, "url.txt")
	if err := os.WriteFile(urlFile, []byte(strings.TrimPrefix(strings.TrimPrefix(s.URL, "https://"), "http://")), 0o644); err != nil {
		return "", err
	}
	out := filepath.Join(workDir, "bumper.mp4")
	var stderr bytes.Buffer
	cmd := exec.CommandContext(ctx, "ffmpeg", buildFFmpegArgs(s, qrPath, captionFiles, urlFile, out)...)
	cmd.WaitDelay = 5 * time.Second
	cmd.Stderr = &stderr
	if err := cmd.Run(); err != nil {
		return "", fmt.Errorf("ffmpeg bumper: %w: %s", err, tail(stderr.String(), 600))
	}
	return out, nil
}

// buildFFmpegArgs is the ffmpeg command line: pure, so tests can assert it.
// Inputs: 0 = background (image or colour), 1 = QR, 2 = audio (music or
// silence). Video: H.264 High 1080p30 yuv420p, keyframes every 3 s so every
// HLS segment is 3 s (well under hls.TargetDuration); audio AAC 160k.
func buildFFmpegArgs(s Spec, qrPath string, captionFiles []string, urlFile, outPath string) []string {
	secs := strconv.Itoa(s.Seconds)
	fps := strconv.Itoa(s.FPS)
	args := []string{"-y"}
	if s.BackgroundImage != "" {
		args = append(args, "-loop", "1", "-t", secs, "-i", s.BackgroundImage)
	} else {
		args = append(args, "-f", "lavfi", "-i", fmt.Sprintf("color=c=%s:s=1920x1080:r=%s:d=%s", s.Background, fps, secs))
	}
	args = append(args, "-i", qrPath)
	if s.Music != "" {
		args = append(args, "-stream_loop", "-1", "-i", s.Music)
	} else {
		args = append(args, "-f", "lavfi", "-i", "anullsrc=r=48000:cl=stereo")
	}

	var g []string
	g = append(g, fmt.Sprintf("[0:v]scale=1920:1080:force_original_aspect_ratio=increase,crop=1920:1080,setsar=1,fps=%s,format=yuv420p[bg]", fps))
	g = append(g, fmt.Sprintf("[1:v]scale=%d:%d,pad=%d:%d:%d:%d:color=white[qr]", qrSize, qrSize, qrCard, qrCard, (qrCard-qrSize)/2, (qrCard-qrSize)/2))
	// The card sits a little below centre; the caption lines go above it,
	// the URL under it.
	g = append(g, fmt.Sprintf("[bg][qr]overlay=(W-w)/2:(H-h)/2+70[v0]"))
	last := "[v0]"
	text := ""
	for i, f := range captionFiles {
		text += fmt.Sprintf("drawtext=fontfile='%s':textfile='%s':fontsize=58:fontcolor=white:x=(w-tw)/2:y=%d,", s.FontPath, f, 96+i*76)
	}
	text += fmt.Sprintf("drawtext=fontfile='%s':textfile='%s':fontsize=40:fontcolor=0xDDDDDD:x=(w-tw)/2:y=h-110,", s.FontPath, urlFile)
	text += fmt.Sprintf("fade=t=in:st=0:d=0.5,fade=t=out:st=%s:d=0.5", trimFloat(float64(s.Seconds)-0.5))
	g = append(g, last+text+"[outv]")
	args = append(args, "-filter_complex", strings.Join(g, ";"), "-map", "[outv]", "-map", "2:a")

	if s.Music != "" {
		args = append(args, "-af", fmt.Sprintf("volume=0.5,afade=t=in:d=1,afade=t=out:st=%d:d=1", s.Seconds-1), "-shortest")
	}
	args = append(args,
		"-t", secs,
		"-c:a", "aac", "-b:a", "160k",
		"-r", fps,
		"-c:v", "libx264", "-pix_fmt", "yuv420p", "-preset", "veryfast", "-crf", "20",
		"-g", strconv.Itoa(3*s.FPS), "-keyint_min", strconv.Itoa(3*s.FPS), "-sc_threshold", "0",
		"-movflags", "+faststart",
		outPath,
	)
	return args
}

func trimFloat(f float64) string { return strconv.FormatFloat(f, 'f', -1, 64) }

func tail(s string, n int) string {
	if len(s) <= n {
		return s
	}
	return s[len(s)-n:]
}
```

```go
// Command make-bumper renders the QR bumper (internal/bumper), segments it
// for HLS, uploads it under hls/bumpers/<hash8>/ and registers it in the
// bumpers table. The newest row is what new lineup versions air, so running
// this is how the bumper is changed. Rerunning with the same inputs is a
// no-op (the content hash exists) and prints the existing id.
//
// Reads DATABASE_URL, BUNNY_* and VIDEO_FONT_PATH from the environment.
// With -render-only PATH it only writes the MP4 there (no DB, no upload):
// handy to eyeball the result locally.
package main

import (
	"context"
	"flag"
	"fmt"
	"log"
	"os"
	"os/signal"
	"path/filepath"
	"syscall"
	"time"

	"github.com/dwellingtw/backend/internal/bumper"
	"github.com/dwellingtw/backend/internal/bunny"
	"github.com/dwellingtw/backend/internal/config"
	"github.com/dwellingtw/backend/internal/db"
	"github.com/dwellingtw/backend/internal/hls"
	"github.com/dwellingtw/backend/internal/linear"
)

const toolMaxConns = 2

func main() {
	spec := bumper.Spec{FPS: config.MinVideoFPS}
	flag.StringVar(&spec.URL, "url", bumper.DefaultURL, "URL the QR encodes")
	flag.StringVar(&spec.Caption, "caption", bumper.DefaultCaption, `caption; "\n" separates lines`)
	flag.IntVar(&spec.Seconds, "seconds", bumper.DefaultSeconds, "bumper length")
	flag.StringVar(&spec.Background, "bg", bumper.DefaultColor, "background colour (ffmpeg syntax)")
	flag.StringVar(&spec.BackgroundImage, "bg-image", "", "background image (16:9), overrides -bg")
	flag.StringVar(&spec.Music, "music", "", "music file, e.g. assets/music/03-calm-piano.mp3 (silent without)")
	name := flag.String("name", "default", "label stored with the bumper")
	renderOnly := flag.String("render-only", "", "write the MP4 here and stop (no upload, no database)")
	uploadConcurrency := flag.Int("upload-concurrency", 8, "segment uploads in parallel")
	flag.Parse()
	spec.FontPath = envStr("VIDEO_FONT_PATH", "/usr/share/fonts/dejavu/DejaVuSans-Bold.ttf")
	if v := os.Getenv("VIDEO_FPS"); v != "" {
		fmt.Sscanf(v, "%d", &spec.FPS)
	}
	spec.Caption = unescapeNewlines(spec.Caption)

	ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
	defer stop()
	if err := run(ctx, spec, *name, *renderOnly, *uploadConcurrency); err != nil {
		log.Fatal(err)
	}
}

func run(ctx context.Context, spec bumper.Spec, name, renderOnly string, uploadConcurrency int) error {
	if spec.Seconds < 3 || spec.Seconds > 60 {
		return fmt.Errorf("-seconds must be 3..60, got %d", spec.Seconds)
	}
	if spec.FPS < config.MinVideoFPS {
		spec.FPS = config.MinVideoFPS
	}
	workDir, err := os.MkdirTemp("", "make-bumper-")
	if err != nil {
		return err
	}
	defer os.RemoveAll(workDir)

	if renderOnly != "" {
		mp4, err := bumper.Render(ctx, spec, workDir)
		if err != nil {
			return err
		}
		if err := os.Rename(mp4, renderOnly); err != nil {
			// Across filesystems rename fails; copy instead.
			data, rerr := os.ReadFile(mp4)
			if rerr != nil {
				return rerr
			}
			if err := os.WriteFile(renderOnly, data, 0o644); err != nil {
				return err
			}
		}
		log.Printf("rendered %s", renderOnly)
		return nil
	}

	hash, err := spec.Hash()
	if err != nil {
		return fmt.Errorf("hash inputs: %w", err)
	}
	pool, err := db.Connect(ctx, os.Getenv("DATABASE_URL"), toolMaxConns)
	if err != nil {
		return err
	}
	defer pool.Close()
	repo := linear.NewRepository(pool)
	if existing, err := repo.BumperByHash(ctx, hash); err != nil {
		return err
	} else if existing != nil {
		log.Printf("bumper with these inputs already exists: id=%d name=%s created=%s", existing.ID, existing.Name, existing.CreatedAt.Format(time.RFC3339))
		fmt.Println(existing.ID)
		return nil
	}

	mp4, err := bumper.Render(ctx, spec, workDir)
	if err != nil {
		return err
	}
	dir := filepath.Join(workDir, "hls")
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return err
	}
	clip, err := hls.NewSegmenter().Segment(ctx, mp4, dir)
	if err != nil {
		return err
	}
	up := bunny.New(os.Getenv("BUNNY_STORAGE_ZONE"), os.Getenv("BUNNY_API_KEY"),
		envStr("BUNNY_STORAGE_HOST", "storage.bunnycdn.com"), os.Getenv("BUNNY_CDN_BASE_URL"), 300*time.Second)
	base, err := hls.Upload(ctx, up, dir, hls.BumperPrefix(hash), clip, max(uploadConcurrency, 1))
	if err != nil {
		return err
	}
	id, created, err := repo.InsertBumper(ctx, &linear.Bumper{Name: name, ContentHash: hash, BaseURL: base, SegmentMS: clip.SegmentMS, TotalMS: clip.TotalMS})
	if err != nil {
		return err
	}
	log.Printf("bumper id=%d created=%v: %d segments, %d ms → %s", id, created, len(clip.SegmentMS), clip.TotalMS, base)
	log.Printf("new lineup versions will carry it within LINEAR_LINEUP_HOURS")
	fmt.Println(id)
	return nil
}

func unescapeNewlines(s string) string {
	out := make([]byte, 0, len(s))
	for i := 0; i < len(s); i++ {
		if s[i] == '\\' && i+1 < len(s) && s[i+1] == 'n' {
			out = append(out, '\n')
			i++
			continue
		}
		out = append(out, s[i])
	}
	return string(out)
}

func envStr(key, def string) string {
	if v := os.Getenv(key); v != "" {
		return v
	}
	return def
}
```

```go
// BumperPrefix is the storage directory of a bumper: hls/bumpers/<hash8>.
func BumperPrefix(contentHash string) string {
	h := contentHash
	if len(h) > 8 {
		h = h[:8]
	}
	return path.Join("hls", "bumpers", h)
}
```
